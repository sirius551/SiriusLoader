-- 1) Если есть зарегистрированная функция выгрузки (с этого или предыдущего запуска) — вызываем её.
if getgenv then
    if getgenv().HubUnload then pcall(getgenv().HubUnload) end
end
-- 2) Подчищаем любые "осиротевшие" интерфейсы по имени — на случай, если прошлый
--    запуск упал ДО регистрации HubUnload (старая версия скрипта, краш на старте и т.п.)
--    и обычный pcall(getgenv().HubUnload) выше их не поймал.
do
    -- Каждый root проверяем и чистим сразу по месту (без промежуточного массива
    -- с возможными "дырами" из nil, которые обрывают ipairs раньше времени).
    local function sweepRoot(root)
        if not root then return end
        local ok, children = pcall(function() return root:GetChildren() end)
        if not ok or not children then return end
        for _, child in ipairs(children) do
            if child.Name == "SiriusLoaderHub" then
                pcall(function() child:Destroy() end)
            end
        end
    end
    pcall(function() sweepRoot(game:GetService("CoreGui")) end)
    pcall(function() if gethui then sweepRoot(gethui()) end end)
    pcall(function() sweepRoot(Players.LocalPlayer:WaitForChild("PlayerGui", 2)) end)
end

local Conns, Cleanups, AccentObjects = {}, {}, {}

local function Add(c) table.insert(Conns, c) return c end
local function OnUnload(fn) table.insert(Cleanups, fn) end

local function New(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

local function Round(o, r) return New("UICorner", {CornerRadius = UDim.new(0, r or 6)}, o) end

local function Tween(o, t, props, style, dir)
    local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function BindAccent(inst, prop)
    table.insert(AccentObjects, {inst, prop})
    inst[prop] = Theme.Accent
end

local function GetChar() return LocalPlayer.Character end
local function GetHum()
    local c = GetChar()
    return c and c:FindFirstChildOfClass("Humanoid")
end
local function GetRoot()
    local c = GetChar()
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function nearestEnemy(maxDist, teamCheck)
    local root = GetRoot()
    if not root then return nil end
    local best, bd, bhum = nil, maxDist or math.huge, nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and not (teamCheck and p.Team ~= nil and p.Team == LocalPlayer.Team) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local tr = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and tr then
                local d = (tr.Position - root.Position).Magnitude
                if d < bd then bd, best, bhum = d, p, hum end
            end
        end
    end
    return best, bd, bhum
end


-- Безопасный клик: на мобильных (Delta) mouse1click может крашить, поэтому там используем Tool:Activate / VirtualUser
local function ClickOnce()
    local c = LocalPlayer.Character
    local tool = c and c:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() tool:Activate() end)
        return
    end
    if not isMobile and mouse1click then
        pcall(mouse1click)
        return
    end
    pcall(function()
        local vu   = game:GetService("VirtualUser")
        local cam  = workspace.CurrentCamera

    end)
end

local function BindBg(inst, prop, darkVal, lightVal)
    local entry = {inst, prop, darkVal, lightVal}
    table.insert(BgObjects, entry)
    inst[prop] = isLightTheme and lightVal or darkVal
end

local function ApplyFullTheme(light)
    isLightTheme = light
    local t = light and THEMES.light or THEMES.dark
    Theme.Background, Theme.Top = t.Background, t.Top
    Theme.Sidebar, Theme.Element = t.Sidebar, t.Element
    Theme.ElementHover, Theme.Text = t.ElementHover, t.Text
    Theme.SubText = t.SubText
    for i = #BgObjects, 1, -1 do
        local e = BgObjects[i]
        if e[1] and e[1].Parent then
            e[1][e[2]] = light and e[4] or e[3]
        else
            table.remove(BgObjects, i)
        end
    end

    -- Repaint neutral UI surfaces and text, including controls created with literal colors.
    if ScreenGui and ScreenGui.Parent then
        for _, obj in ipairs(ScreenGui:GetDescendants()) do
            if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                if obj.BackgroundTransparency >= 0.85 then
                    obj.TextColor3 = light and t.Text or Color3.fromRGB(235, 240, 250)
                else
                    local bg = obj.BackgroundColor3
                    local bgLum = (bg.R * 0.299 + bg.G * 0.587 + bg.B * 0.114)
                    obj.TextColor3 = (light and bgLum > 0.58) and t.Text
                        or ((not light and bgLum < 0.35) and Color3.fromRGB(240, 245, 255) or obj.TextColor3)
                end
            end
            if obj:IsA("GuiObject") and obj.BackgroundTransparency < 0.95 then
                local c = obj.BackgroundColor3
                local hi = math.max(c.R, c.G, c.B)
                local lo = math.min(c.R, c.G, c.B)
                local saturation = hi - lo
                if saturation < 0.16 then
                    obj.BackgroundColor3 = light and THEMES.light.Element or THEMES.dark.Element
                    if obj:IsA("TextButton") or obj:IsA("TextBox") then
                        obj.TextColor3 = light and THEMES.light.Text or THEMES.dark.Text
                    end
                end
            end
        end
    end

    -- Make light mode intentionally opaque and remove the game scene from behind controls.
    if MainFrame then
        local transparency = math.clamp(tonumber(Config.Transparency) or 0, 0, 0.85)
        MainFrame.BackgroundTransparency = transparency
        WallpaperTint.BackgroundColor3 = light and Color3.fromRGB(242, 245, 251) or THEMES.dark.Background
        WallpaperTint.BackgroundTransparency = math.clamp(0.42 + transparency * 0.5, 0.42, 0.9)
        WallpaperImage.ImageTransparency = light and 1 or math.clamp(tonumber(Config.WallpaperTransparency) or 0.15, 0, 0.9)
        TopBar.BackgroundTransparency = light and 0 or 0.12
        Sidebar.BackgroundTransparency = light and 0 or 0.12
        SearchBox.BackgroundTransparency = light and 0 or 0.08
    end
end

local function NormalizeAssetId(value)
    value = tostring(value or ""):match("^%s*(.-)%s*$")
    if value == "" then return "" end
    local digits = value:match("(%d+)")
    return digits and ("rbxassetid://" .. digits) or ""
end

local wallpaperFrameIndex, wallpaperClock = 0, 0
local function ApplyWallpaper()
    local spriteId = NormalizeAssetId(Config.WallpaperSpriteId)
    local staticId = NormalizeAssetId(Config.WallpaperId)
    local filePath = tostring(Config.WallpaperFilePath or "")
    local localAsset = ""
    if filePath ~= "" then
        local getter = getcustomasset or getsynasset
        if getter then
            local ok, asset = pcall(getter, filePath)
            if ok and type(asset) == "string" then localAsset = asset end
        end
    end
    WallpaperImage.Image = localAsset ~= "" and localAsset or (spriteId ~= "" and spriteId or staticId)
    WallpaperImage.ImageTransparency = math.clamp(tonumber(Config.WallpaperTransparency) or 0.15, 0, 0.9)
    wallpaperFrameIndex, wallpaperClock = 0, 0
    if spriteId ~= "" or (localAsset ~= "" and Config.WallpaperFileAnimated == true) then
        local cols = math.clamp(math.floor(tonumber(Config.WallpaperColumns) or 4), 1, 16)
        local frames = math.clamp(math.floor(tonumber(Config.WallpaperFrames) or 16), 1, 256)
        local side = 1024 / cols
        WallpaperImage.ImageRectSize = Vector2.new(side, side)
        WallpaperImage.ImageRectOffset = Vector2.zero
        Config.WallpaperColumns, Config.WallpaperFrames = cols, frames
    else
        WallpaperImage.ImageRectSize = Vector2.zero
        WallpaperImage.ImageRectOffset = Vector2.zero
    end
end
ApplyWallpaper()
Add(RunService.Heartbeat:Connect(function(dt)
    if not WallpaperImage.Parent then return end
    local hasSprite = NormalizeAssetId(Config.WallpaperSpriteId) ~= "" or (tostring(Config.WallpaperFilePath or "") ~= "" and Config.WallpaperFileAnimated == true)
    if not hasSprite then return end
    local fps = math.clamp(tonumber(Config.WallpaperFPS) or 12, 1, 30)
    wallpaperClock += dt
    if wallpaperClock >= 1 / fps then
        wallpaperClock %= 1 / fps
        local frames = math.clamp(math.floor(tonumber(Config.WallpaperFrames) or 16), 1, 256)
        local cols = math.clamp(math.floor(tonumber(Config.WallpaperColumns) or 4), 1, 16)
        wallpaperFrameIndex = (wallpaperFrameIndex + 1) % frames
        WallpaperImage.ImageRectOffset = Vector2.new((wallpaperFrameIndex % cols) * (1024 / cols), math.floor(wallpaperFrameIndex / cols) * (1024 / cols))

-- Перетаскивание: координаты всегда считаются в пикселях экрана.
-- Окно остаётся целиком видимым, а верхняя панель не уходит под системную шторку.
local SAFE_TOP    = 60
local SAFE_BOTTOM = 12
local SAFE_SIDE   = 8

local function ClampFrameToScreen(frame)
    local camera = workspace.CurrentCamera
    if not camera or not frame or not frame.Parent then return end

    local vp = camera.ViewportSize
    local size = frame.AbsoluteSize
    local pos = frame.AbsolutePosition
    local minX, minY = SAFE_SIDE, SAFE_TOP
    local maxX = math.max(minX, vp.X - size.X - SAFE_SIDE)
    local maxY = math.max(minY, vp.Y - size.Y - SAFE_BOTTOM)
    local left = math.clamp(pos.X, minX, maxX)
    local top = math.clamp(pos.Y, minY, maxY)

    if math.abs(left - pos.X) > 0.5 or math.abs(top - pos.Y) > 0.5 then
        local parent = frame.Parent
        local parentPos, parentSize = Vector2.zero, vp
        if parent:IsA("GuiObject") then
            parentPos, parentSize = parent.AbsolutePosition, parent.AbsoluteSize
        end
        local uiScale = frame:IsDescendantOf(MainFrame) and math.max(MainScale.Scale, 0.05) or 1
        local p = frame.Position
        local anchorX = size.X * frame.AnchorPoint.X
        local anchorY = size.Y * frame.AnchorPoint.Y
        local offsetX = (left + anchorX - parentPos.X - parentSize.X * p.X.Scale) / uiScale
        local offsetY = (top + anchorY - parentPos.Y - parentSize.Y * p.Y.Scale) / uiScale
        frame.Position = UDim2.new(p.X.Scale, offsetX, p.Y.Scale, offsetY)
    end
end

local function MakeDraggable(handle, target)
    local dragging, dragInput, dragStart, startPosition
    handle.Active = true
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            if UIS:GetFocusedTextBox() then return end
            dragging = true
            dragInput = input
            dragStart = input.Position
            startPosition = target.Position
        end
    end)

    Add(UIS.InputChanged:Connect(function(input)
        if not dragging then return end
        local isMouseMove = input.UserInputType == Enum.UserInputType.MouseMovement
        local isTouchMove = input.UserInputType == Enum.UserInputType.Touch
        if not isMouseMove and not isTouchMove then return end
        local delta = input.Position - dragStart
        local p = startPosition
        target.Position = UDim2.new(p.X.Scale, p.X.Offset + delta.X, p.Y.Scale, p.Y.Offset + delta.Y)
    end))

    Add(UIS.InputEnded:Connect(function(input)
        if not dragging then return end
        if input == dragInput
        or input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            dragInput, dragStart, startPosition = nil, nil, nil
            task.defer(function() ClampFrameToScreen(target) end)
        end
    end))
end

-- Список сохранённых профилей: { name = "...", file = "..." }
local ProfileList   = {}
local ActiveProfile = nil  -- имя текущего активного профиля

local function ProfilePath(name)
    return PROFILES_DIR .. name:gsub("[^%w_%-]", "_") .. ".json"
end

local function EnsureProfilesDir()
    pcall(function()
        if makefolder and not isfolder(PROFILES_DIR) then
            makefolder(PROFILES_DIR)
        end
    end)
end

local function LoadProfileIndex()
    pcall(function()
        if isfile and readfile and isfile(PROFILE_INDEX) then
            local data = HttpService:JSONDecode(readfile(PROFILE_INDEX))
            if type(data) == "table" then
                ProfileList   = data.profiles or {}
                ActiveProfile = data.active or nil
            end
        end
    end)
end

local function SaveProfileIndex()
    pcall(function()
        if writefile then
            writefile(PROFILE_INDEX, HttpService:JSONEncode({
                profiles = ProfileList,
                active   = ActiveProfile,
            }))
        end
    end)
end

-- Собирает текущее состояние всех зарегистрированных виджетов
local function SnapshotState()
    local snap = {}
    for key, obj in pairs(WidgetRegistry) do
        local ok, v = pcall(function() return obj:Get() end)
        if ok then snap[key] = v end
    end
    return snap
end

-- Применяет снапшот к виджетам (silent = не вызывать callbacks)
local function ApplySnapshot(snap, silent)
    for key, v in pairs(snap) do
        local obj = WidgetRegistry[key]
        if obj then
            pcall(function()
                obj:Set(v, silent)
            end)
        end
    end
end

local function SaveProfile(name)
    EnsureProfilesDir()
    local snap = SnapshotState()
    local path = ProfilePath(name)
    pcall(function()
        if writefile then
            writefile(path, HttpService:JSONEncode(snap))
        end
    end)
    -- Обновляем индекс
    local found = false
    for _, p in ipairs(ProfileList) do
        if p.name == name then found = true break end
    end
    if not found then
        table.insert(ProfileList, {name = name, file = path})
    end
    ActiveProfile = name
    SaveProfileIndex()
    Notify("Конфиги", "Сохранено: " .. name, 2)
end

local function LoadProfile(name)
    local path = ProfilePath(name)
    local ok, data = pcall(function()
        if isfile and readfile and isfile(path) then
            return HttpService:JSONDecode(readfile(path))
        end
    end)
    if ok and type(data) == "table" then
        ApplySnapshot(data, false)
        ActiveProfile = name
        SaveProfileIndex()
        Notify("Конфиги", "Загружено: " .. name, 2)
        return true
    else
        Notify("Конфиги", "Не найден: " .. name, 2)
        return false
    end
end

local function DeleteProfile(name)
    local path = ProfilePath(name)
    pcall(function() if delfile then delfile(path) end end)
    for i, p in ipairs(ProfileList) do
        if p.name == name then table.remove(ProfileList, i) break end
    end
    if ActiveProfile == name then ActiveProfile = nil end
    SaveProfileIndex()
    Notify("Конфиги", "Удалён: " .. name, 2)
end

local function RenameProfile(oldName, newName)
    if oldName == newName then return end
    local oldPath = ProfilePath(oldName)
    local newPath = ProfilePath(newName)
    pcall(function()
        if isfile and readfile and writefile and isfile(oldPath) then
            writefile(newPath, readfile(oldPath))
            if delfile then delfile(oldPath) end
        end
    end)
    for _, p in ipairs(ProfileList) do
        if p.name == oldName then
            p.name = newName
            p.file = newPath
            break
        end
    end
    if ActiveProfile == oldName then ActiveProfile = newName end
    SaveProfileIndex()
    Notify("Конфиги", oldName .. " → " .. newName, 2)
end

local function FetchSource(url)
    local src
    local ok = pcall(function() src = game:HttpGet(url) end)
    if ok and type(src) == "string" and #src > 0 then return src end
    ok = pcall(function() src = game:HttpGet(url, true) end)
    if ok and type(src) == "string" and #src > 0 then return src end
    local req = (syn and syn.request) or (http and http.request) or http_request or request
    if req then
        local ok2, res = pcall(req, {Url = url, Method = "GET"})
        if ok2 and res and type(res.Body) == "string" and #res.Body > 0 then return res.Body end
    end
    return nil
end

-- Подменяет loadstring внутри запускаемого кода: если скачанный текст не Lua (404, HTML и т.д.),
-- вместо «attempt to call a nil value» будет настоящая причина и начало ответа сервера.
-- Дочерние скрипты Sirius: каждый получает собственный контейнер очистки.
local ChildScripts = {}
local ChildScriptId = 0

local function CleanupChild(child)
    if not child or child.Stopped then return end
    child.Stopped = true
    for i = #child.Cleanups, 1, -1 do pcall(child.Cleanups[i]) end
    for _, connection in ipairs(child.Connections) do
        pcall(function() connection:Disconnect() end)
    end
    for _, instance in ipairs(child.Instances) do
        pcall(function() instance:Destroy() end)
    end
    child.Cleanups, child.Connections, child.Instances = {}, {}, {}
end

local function CleanupChildScripts()
    for i = #ChildScripts, 1, -1 do CleanupChild(ChildScripts[i]) end
    table.clear(ChildScripts)
end

local function CreateChildScript(name)
    ChildScriptId += 1
    local child = {
        Id = ChildScriptId, Name = tostring(name), Stopped = false,
        Cleanups = {}, Connections = {}, Instances = {},
    }
    function child:IsRunning() return not self.Stopped end
    function child:AddCleanup(callback)
        if type(callback) ~= "function" then return false end
        if self.Stopped then pcall(callback) else table.insert(self.Cleanups, callback) end
        return true
    end
    function child:TrackConnection(connection)
        if typeof(connection) == "RBXScriptConnection" then
            if self.Stopped then pcall(function() connection:Disconnect() end)
            else table.insert(self.Connections, connection) end
        end
        return connection
    end
    function child:TrackInstance(instance)
        if typeof(instance) == "Instance" then
            if self.Stopped then pcall(function() instance:Destroy() end)
            else table.insert(self.Instances, instance) end
        end
        return instance
    end
    function child:Spawn(callback)
        if type(callback) ~= "function" or self.Stopped then return nil end
        local thread = task.spawn(function()
            if not self.Stopped then
                local ok, err = pcall(callback, self)
                if not ok then warn("[Sirius child: " .. self.Name .. "] " .. tostring(err)) end
            end
        end)
        self:AddCleanup(function() pcall(task.cancel, thread) end)
        return thread
    end
    table.insert(ChildScripts, child)
    return child
end

local function wrapEnv(fn, child)
    pcall(function()
        local base = (getgenv and getgenv()) or getfenv(0)
        local env = setmetatable({
            SiriusChild = child,
            loadstring = function(src, ...)
                local f, e = loadstring(src, ...)
                if not f then
                    local head = string.gsub(string.sub(tostring(src), 1, 80), "%s+", " ")
                    error("loadstring не смог скомпилировать код: " .. tostring(e) .. " | начало ответа: " .. head, 0)
                end
                return f
            end,
        }, {__index = base, __newindex = base})
        setfenv(fn, env)
    end)
    return fn
end
